get the DB wallet (you need your .oci/config etc setup)
oci db autonomous-database generate-wallet \
  --autonomous-database-id <autonomous_database_ocid> \
  --password '<wallet_password>' \
  --generate-type SINGLE \
  --file wallet.zip
  
unzip the file into a folder, checn the tns_names.ora file in there to get the connection name (e.g. telemetry_high)

Upcate the config properties for the timeseries DB to use these
timeseriesdb.jdbc.connectionname=<connection name>
timeseriesdb.jdbc.walletpath=<wallet path - e.g. ./configsecure/timeseriesdb>
timeseriesdb.jdbc.username=<uname of ingest user>
timeseriesdb.jdbc.password=<pw of ingest user>
